cmake_minimum_required(VERSION 3.20)
project(candle_lfm2_moe_probe LANGUAGES C CXX)
if(NOT DEFINED LLAMA_SOURCE)
  message(FATAL_ERROR "LLAMA_SOURCE must select the locked 582eab71 source tree")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
add_subdirectory("${LLAMA_SOURCE}" llama)
add_executable(candle-lfm2-moe-reference llama_probe.cpp)
target_compile_features(candle-lfm2-moe-reference PRIVATE cxx_std_17)
target_include_directories(candle-lfm2-moe-reference PRIVATE "${LLAMA_SOURCE}/vendor/nlohmann")
target_link_libraries(candle-lfm2-moe-reference PRIVATE llama)
